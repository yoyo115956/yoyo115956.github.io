## 2.2 Python 解析脚本

```python
#!/usr/bin/env python3
"""解析 gem5 stats.txt 文件"""

import re
import os

def parse_stats(stats_file):
    """从 stats.txt 中提取指定指标"""
    stats = {}
    with open(stats_file, 'r') as f:
        for line in f:
            # 跳过注释和空行
            if line.startswith('#') or line.strip() == '':
                continue
            # 匹配 "key    value" 格式
            match = re.match(r'^(\S+)\s+([\d.eE+\-naninf]+)', line)
            if match:
                key, val = match.group(1), match.group(2)
                try:
                    stats[key] = float(val)
                except ValueError:
                    stats[key] = val
    return stats

def get_ipc(stats_file):
    stats = parse_stats(stats_file)
    return stats.get('system.cpu.ipc', 0)

def get_cache_miss_rate(stats_file, level='l1d'):
    stats = parse_stats(stats_file)
    key_map = {
        'l1i': 'system.cpu.icache.overallMissRate::total',
        'l1d': 'system.cpu.dcache.overallMissRate::total',
        'l2':  'system.l2cache.overallMissRate::total',
    }
    return stats.get(key_map[level], 0)
```

# 三、参数扫描实验

## 3.1 研究问题示例：L1 DCache 大小对 IPC 的影响

```python
#!/usr/bin/env python3
"""批量运行 gem5 参数扫描实验"""

import subprocess
import os
import json

GEM5_BIN = "build/X86/gem5.opt"
CONFIG = "configs/example/se.py"
WORKLOAD = "./benchmark_array"

# 实验参数矩阵
cache_sizes = ['8kB', '16kB', '32kB', '64kB', '128kB', '256kB']
benchmarks = ['matrix_mul', 'linked_list', 'fft']

results = {}

for bench in benchmarks:
    results[bench] = {}
    for size in cache_sizes:
        outdir = f"results/{bench}_dcache_{size}"
        os.makedirs(outdir, exist_ok=True)
        
        cmd = [
            GEM5_BIN, f"--outdir={outdir}",
            CONFIG,
            f"--cmd=./{bench}",
            "--cpu-type=DerivO3CPU",
            "--caches", "--l2cache",
            f"--l1d_size={size}",
            "--l1d_assoc=8",
            "--l1i_size=32kB",
            "--l2_size=4MB",
            "--maxinsts=100000000"
        ]
        
        print(f"Running: {bench} with L1D={size}")
        subprocess.run(cmd, stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        
        stats_file = f"{outdir}/stats.txt"
        results[bench][size] = {
            'ipc': get_ipc(stats_file),
            'l1d_miss_rate': get_cache_miss_rate(stats_file, 'l1d'),
            'l2_miss_rate': get_cache_miss_rate(stats_file, 'l2'),
        }

# 保存结果
with open('sweep_results.json', 'w') as f:
    json.dump(results, f, indent=2)
print("Done! Results saved to sweep_results.json")
```

## 3.2 结果可视化

```python
#!/usr/bin/env python3
"""可视化参数扫描结果"""

import json
import matplotlib.pyplot as plt
import numpy as np

with open('sweep_results.json') as f:
    results = json.load(f)

cache_sizes = ['8kB', '16kB', '32kB', '64kB', '128kB', '256kB']
benchmarks = list(results.keys())
x = np.arange(len(cache_sizes))

fig, axes = plt.subplots(1, 2, figsize=(14, 5))

# 图1：IPC vs Cache Size
ax = axes[0]
for bench in benchmarks:
    ipcs = [results[bench][s]['ipc'] for s in cache_sizes]
    ax.plot(x, ipcs, marker='o', label=bench)
ax.set_xticks(x)
ax.set_xticklabels(cache_sizes, rotation=30)
ax.set_xlabel('L1 DCache Size')
ax.set_ylabel('IPC')
ax.set_title('IPC vs L1 DCache Size')
ax.legend()
ax.grid(True, alpha=0.3)

# 图2：L1D Miss Rate vs Cache Size
ax = axes[1]
for bench in benchmarks:
    mrs = [results[bench][s]['l1d_miss_rate'] * 100 for s in cache_sizes]
    ax.plot(x, mrs, marker='s', label=bench)
ax.set_xticks(x)
ax.set_xticklabels(cache_sizes, rotation=30)
ax.set_xlabel('L1 DCache Size')
ax.set_ylabel('Miss Rate (%)')
ax.set_title('L1D Cache Miss Rate vs Size')
ax.legend()
ax.grid(True, alpha=0.3)

plt.tight_layout()
plt.savefig('cache_sweep.pdf', dpi=150, bbox_inches='tight')
plt.show()
print("Figure saved to cache_sweep.pdf")
```

# 四、ROI（Region of Interest）仿真

对于长程序，只仿真**感兴趣的代码段**，节省时间：

```c
// 在 C 程序中插入 gem5 magic instruction
#include "m5ops.h"

int main() {
    // 初始化阶段（Atomic 快速跳过）
    initialize_data();
    
    m5_reset_stats(0, 0);  // 重置统计计数器，开始 ROI
    
    // 核心计算（需要精确仿真的部分）
    for (int i = 0; i < N; i++) {
        compute_intensive(data[i]);
    }
    
    m5_dump_stats(0, 0);   // 导出统计数据，结束 ROI
    
    return 0;
}
```

# 五、Checkpoint 工作流

对于需要引导 OS 的 FS 模式仿真，Checkpoint 至关重要：

```bash
# 第一步：AtomicSimple 快速运行到目标指令数，保存 checkpoint
build/X86/gem5.opt configs/example/se.py \
    --cpu-type=AtomicSimpleCPU \
    --cmd=./spec_benchmark \
    --take-checkpoints=1000000000,1  # 10亿条指令处保存

# 第二步：从 checkpoint 恢复，用 O3CPU 仿真
build/X86/gem5.opt configs/example/se.py \
    --cpu-type=DerivO3CPU \
    --caches --l2cache \
    --cmd=./spec_benchmark \
    --restore-from=m5out/cpt.1000000000/ \
    --maxinsts=100000000
```

# 六、常见问题排查

| 问题 | 可能原因 | 解决方法 |
|------|----------|----------|
| `Segfault in guest` | 程序需要系统调用未实现 | 用 SE 模式或用静态编译 |
| IPC 异常低 | Cache 配置太小，或程序访存模式差 | 检查 miss rate，调整 Cache 配置 |
| 仿真极慢 | O3CPU + 大 Cache + 复杂程序 | 先用 Atomic 做 checkpoint，缩短 ROI |
| stats 全为 0 | 程序运行失败或 maxinsts 太小 | 检查程序输出，去掉 maxinsts 限制 |
| `panic: x87` | 程序使用了 x87 浮点，gem5 未启用 | 编译时加 `-msse2 -mfpmath=sse` |
