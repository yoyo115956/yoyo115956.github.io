> 实现一个支持 RV32I 子集的单周期处理器，是理解处理器微架构的最佳实践。

# 一、总体架构

单周期处理器在**一个时钟周期内完成一条指令**，数据通路（Datapath）与控制逻辑（Control）分离设计。

```
┌──────────────────────────────────────────────────────┐
│                    单周期 RV32I 处理器                 │
│                                                      │
│  PC ──→ InstrMem ──→ Decode ──→ RegFile              │
│   ↑                     │         ↓    ↓             │
│   │              Control│        RS1  RS2             │
│   │                     ↓         ↓    ↓             │
│   │               ImmGen        ──→ ALU ←──          │
│   │                     │         ↓                  │
│   └──── PCNext ←─────── │      DataMem               │
│                          │         ↓                  │
│                          └──→  WriteBack → RegFile   │
└──────────────────────────────────────────────────────┘
```

# 二、各模块实现

## 2.1 PC 寄存器与取指

```verilog
module pc_reg (
    input  wire        clk, rst,
    input  wire [31:0] pc_next,
    output reg  [31:0] pc
);
    always @(posedge clk or posedge rst)
        if (rst) pc <= 32'h0000_0000;
        else     pc <= pc_next;
endmodule

// PC+4（顺序执行）或 分支/跳转目标
assign pc_next = (branch_taken) ? branch_target :
                 (jal_inst)     ? jal_target     :
                 (jalr_inst)    ? jalr_target     :
                                  pc + 4;
```

## 2.2 立即数生成器（ImmGen）

RISC-V 的 6 种指令格式有不同的立即数编码，需要统一提取并符号扩展：

```verilog
module imm_gen (
    input  wire [31:0] instr,
    output reg  [31:0] imm
);
    wire [6:0] opcode = instr[6:0];
    
    always @(*) begin
        case (opcode)
            7'b0010011, // I-type ALU
            7'b0000011, // Load
            7'b1100111: // JALR
                imm = {{20{instr[31]}}, instr[31:20]};
            
            7'b0100011: // S-type Store
                imm = {{20{instr[31]}}, instr[31:25], instr[11:7]};
            
            7'b1100011: // B-type Branch
                imm = {{19{instr[31]}}, instr[31], instr[7], 
                        instr[30:25], instr[11:8], 1'b0};
            
            7'b0110111, // LUI
            7'b0010111: // AUIPC
                imm = {instr[31:12], 12'b0};
            
            7'b1101111: // JAL
                imm = {{11{instr[31]}}, instr[31], instr[19:12], 
                        instr[20], instr[30:21], 1'b0};
            
            default: imm = 32'b0;
        endcase
    end
endmodule
```

## 2.3 寄存器文件

```verilog
module reg_file (
    input  wire        clk,
    input  wire [4:0]  rs1, rs2, rd,
    input  wire [31:0] wd,
    input  wire        we,
    output wire [31:0] rd1, rd2
);
    reg [31:0] regs [0:31];
    
    // x0 永远为 0
    assign rd1 = (rs1 == 5'b0) ? 32'b0 : regs[rs1];
    assign rd2 = (rs2 == 5'b0) ? 32'b0 : regs[rs2];
    
    always @(posedge clk)
        if (we && rd != 5'b0)
            regs[rd] <= wd;
endmodule
```

## 2.4 ALU

```verilog
module alu (
    input  wire [31:0] a, b,
    input  wire [3:0]  alu_ctrl,
    output reg  [31:0] result,
    output wire        zero
);
    localparam ADD  = 4'd0, SUB  = 4'd1, AND  = 4'd2, OR   = 4'd3;
    localparam XOR  = 4'd4, SLL  = 4'd5, SRL  = 4'd6, SRA  = 4'd7;
    localparam SLT  = 4'd8, SLTU = 4'd9;
    
    always @(*) begin
        case (alu_ctrl)
            ADD:  result = a + b;
            SUB:  result = a - b;
            AND:  result = a & b;
            OR:   result = a | b;
            XOR:  result = a ^ b;
            SLL:  result = a << b[4:0];
            SRL:  result = a >> b[4:0];
            SRA:  result = $signed(a) >>> b[4:0];
            SLT:  result = ($signed(a) < $signed(b)) ? 1 : 0;
            SLTU: result = (a < b) ? 1 : 0;
            default: result = 32'b0;
        endcase
    end
    
    assign zero = (result == 32'b0);
endmodule
```

## 2.5 控制单元

```verilog
module control (
    input  wire [6:0] opcode,
    input  wire [2:0] funct3,
    input  wire [6:0] funct7,
    output reg        reg_write, mem_read, mem_write,
    output reg        mem_to_reg, branch, jal, jalr,
    output reg  [3:0] alu_ctrl,
    output reg        alu_src   // 0=rs2, 1=imm
);
    always @(*) begin
        // 默认值
        {reg_write, mem_read, mem_write, mem_to_reg, 
         branch, jal, jalr, alu_src} = 8'b0;
        alu_ctrl = 4'd0; // ADD
        
        case (opcode)
            7'b0110011: begin // R-type
                reg_write = 1;
                case ({funct7[5], funct3})
                    4'b0000: alu_ctrl = 4'd0; // ADD
                    4'b1000: alu_ctrl = 4'd1; // SUB
                    4'b0111: alu_ctrl = 4'd2; // AND
                    4'b0110: alu_ctrl = 4'd3; // OR
                    4'b0100: alu_ctrl = 4'd4; // XOR
                    4'b0001: alu_ctrl = 4'd5; // SLL
                    4'b0101: alu_ctrl = 4'd6; // SRL
                    4'b1101: alu_ctrl = 4'd7; // SRA
                    4'b0010: alu_ctrl = 4'd8; // SLT
                    4'b0011: alu_ctrl = 4'd9; // SLTU
                endcase
            end
            7'b0010011: begin // I-type ALU
                reg_write = 1; alu_src = 1;
                // (funct3 → alu_ctrl 类似 R-type)
            end
            7'b0000011: begin // Load
                reg_write = 1; mem_read = 1; mem_to_reg = 1; alu_src = 1;
            end
            7'b0100011: begin // Store
                mem_write = 1; alu_src = 1;
            end
            7'b1100011: begin // Branch
                branch = 1;
            end
            7'b1101111: begin // JAL
                jal = 1; reg_write = 1;
            end
            7'b1100111: begin // JALR
                jalr = 1; reg_write = 1; alu_src = 1;
            end
            7'b0110111: begin // LUI
                reg_write = 1; alu_src = 1; alu_ctrl = 4'd0; // result = imm
            end
        endcase
    end
endmodule
```

# 三、仿真验证

## 3.1 测试思路

单周期处理器验证策略：
1. **单指令测试**：每类指令单独验证（ADD/SUB/LW/SW/BEQ...）
2. **程序级测试**：运行排序、斐波那契等小程序
3. **自检程序**：通过 `ecall` 输出结果，对比预期值

## 3.2 Testbench 框架

```verilog
`timescale 1ns/1ps
module tb_cpu;
    reg clk, rst;
    
    // 实例化被测设计
    riscv_cpu dut (
        .clk(clk), .rst(rst)
    );
    
    // 时钟生成
    initial clk = 0;
    always #5 clk = ~clk; // 100 MHz
    
    // 复位 & 运行
    initial begin
        rst = 1;
        #20 rst = 0;
        #10000 $finish;
    end
    
    // 波形导出
    initial begin
        $dumpfile("cpu.vcd");
        $dumpvars(0, tb_cpu);
    end
    
    // 监控寄存器
    always @(posedge clk)
        $display("PC=%h, x1=%d, x2=%d", dut.pc, dut.regs[1], dut.regs[2]);
endmodule
```

# 四、常见错误与调试

| 问题 | 可能原因 |
|------|----------|
| PC 不更新 | rst 信号极性错误，或 pc_next 逻辑 |
| 寄存器写入 x0 | reg_write 未检查 rd==0 |
| 立即数符号扩展错误 | ImmGen 的位拼接顺序 |
| 分支不跳转 | branch 信号逻辑，或 ALU zero 信号 |
| Load 数据错误 | 地址对齐，或字节/半字选择逻辑 |

调试工具推荐：
- **Verilator**：将 Verilog 编译成 C++ 仿真，速度快
- **GTKWave**：查看 VCD 波形
- **Spike**：官方 RISC-V ISA 仿真器，用于对比参考
